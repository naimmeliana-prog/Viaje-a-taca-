<script>
let currentPortal = '';
let currentMac = '';

async function checkPortal() {
const portal = document.getElementById('portal').value.trim();
const mac = document.getElementById('mac').value.trim();
const resultDiv = document.getElementById('result');

if (!portal || !mac) {
resultDiv.innerHTML = `<p class="error">⚠️ Completa ambos campos</p>`;
return;
}

resultDiv.innerHTML = `<p>🔄 Comprobando portal...</p>`;
document.getElementById('btnM3U').disabled = true;

try {
// Nota: Debido a CORS, esta versión solo simula el check.
// Para funcionamiento REAL necesitas un servidor (Python/Flask).
resultDiv.innerHTML = `
<p class="success">✅ Conexión simulada correcta</p>
<p><strong>Portal:</strong> ${portal}</p>
<p><strong>MAC:</strong> ${mac}</p>
<small style="color:#aaa;">(En versión completa se comprobaría realmente)</small>
`;
currentPortal = portal;
currentMac = mac;
document.getElementById('btnM3U').disabled = false;
} catch (e) {
resultDiv.innerHTML = `<p class="error">❌ Error: ${e.message}</p>`;
}
}

async function generateM3U() {
const resultDiv = document.getElementById('result');
resultDiv.innerHTML = `<p>📥 Generando lista M3U...</p>`;

// Simulación de generación
setTimeout(() => {
const m3uContent = `#EXTM3U
#EXTINF:-1,Test Channel 1
\( {currentPortal}/play?mac= \){currentMac.replace(/:/g,'')}&channel=1
#EXTINF:-1,Test Channel 2
\( {currentPortal}/play?mac= \){currentMac.replace(/:/g,'')}&channel=2`;

resultDiv.innerHTML = `
<p class="success">✅ Lista generada (simulada)</p>
<pre>${m3uContent}</pre>
<button onclick="downloadM3U()">⬇️ Descargar playlist.m3u</button>
`;
}, 1200);
}

function downloadM3U() {
const content = `#EXTM3U\n# Playlist generada para MAC: ${currentMac}\n# Portal: \( {currentPortal}\n\n#EXTINF:-1 tvg-id="1",Canal de Prueba\n \){currentPortal}/stream\n`;
const blob = new Blob([content], { type: 'text/plain' });
const url = URL.createObjectURL(blob);
const a = document.createElement('a');
a.href = url;
a.download = 'playlist.m3u';
a.click();
}
